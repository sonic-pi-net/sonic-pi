Can't code yet? Ask an AI

You don't need to know how to code. Ask an AI to write something to get you started. Then change, tweak, mash it up and make it your own.

use_bpm 60
notes = (ring :d4, :a4, :f4, :c5, :e4, :a4, :g4, :d5, :f4, :c5, :a4, :e5)

with_fx :reverb, room: 0.8 do
  # two players, the same twelve notes
  live_loop :left do
    synth :kalimba, note: notes.tick, pan: -0.8, amp: 3
    sleep 0.2
  end

  # this one is a hundredth slower, so it falls behind a note at a time
  live_loop :right do
    synth :kalimba, note: notes.tick, pan: 0.8, amp: 3
    sleep 0.202
  end

  # a third, an octave down at half the speed, holds the middle
  live_loop :centre do
    synth :kalimba, note: notes.tick - 12, amp: 2.5
    sleep 0.4
  end

  live_loop :ground do
    synth :hollow, note: (ring :d2, :a1, :f2, :c2).tick, attack: 4, release: 8, amp: 1.2
    sleep 12
  end

  with_fx :echo, phase: 0.6, decay: 5, mix: 0.5 do
    live_loop :sparks do
      synth :pretty_bell, note: notes.choose + 12, release: 2, amp: 0.25, pan: rrand(-1, 1) if one_in(3)
      sleep 1.2
    end
  end
end

use_bpm 124

live_loop :bar do
  # a new bar every time: how many kicks and hats, and where they fall
  bar = tick
  kicks = spread((ring 3, 5, 4, 5, 3, 7)[bar], 16)
  hats = spread(rrand_i(7, 13), 16).rotate(rrand_i(0, 3))
  snare = spread(2, 16).rotate(4)
  root = (ring :e1, :e1, :g1, :a1)[bar / 2]
  fill = bar % 4 == 3   # every fourth bar ends in a roll

  16.times do |i|
    sample :bd_tek, amp: 2 if kicks[i]
    sample :drum_snare_hard, amp: 0.8 if snare[i]
    sample :drum_snare_hard, amp: 0.3 + i * 0.03, rate: 1.5 if fill and i >= 12
    sample :drum_cymbal_closed, amp: rrand(0.3, 0.7), rate: 1.5 if hats[i]
    sample :drum_cymbal_open, amp: 0.4, finish: 0.3 if i % 4 == 2 and one_in(2)
    # the bass goes wherever the kick does not, opening up through the bar
    synth :tb303, note: root + (ring 0, 12, 7).choose, release: 0.15, cutoff: 60 + i * 4, res: 0.9, amp: 0.7 unless kicks[i]
    sleep 0.25
  end
end

use_bpm 100

# a bass that growls: its filter opens and closes while each note holds
live_loop :bass do
  use_synth :dsaw
  s = play (ring :d1, :d1, :f1, :c1).tick, sustain: 3.5, release: 0.5, detune: 0.2, cutoff: 50, cutoff_slide: 2, amp: 0.6
  control s, cutoff: 95
  sleep 2
  control s, cutoff: 55
  sleep 2
end

# half-time drums through distortion
with_fx :distortion, distort: 0.5 do

    sample :bd_boom, amp: 1
    sample :bd_tek, amp: 0.8
    sleep 1.75
    sample :bd_tek, amp: 0.6
    sleep 0.25
    sample :sn_zome, amp: 0.7
    sleep 2
  end
end

live_loop :hats do
  # mostly sixteenths, now and then a burst twice as fast
  if one_in(6)
    4.times do
      sample :elec_tick, amp: 0.6, rate: 2
      sleep 0.125
    end

# neon: an arpeggio whose brightness sweeps up and down, echoing into the dark
with_fx :reverb, room: 0.9 do
  with_fx :echo, phase: 0.375, decay: 4, mix: 0.4 do
    live_loop :neon do
      use_synth :blade
      notes = scale(:d3, :minor_pentatonic, num_octaves: 3)
      play notes[(ring 0, 2, 4, 7, 4, 2, 9, 7).tick], release: 0.25, cutoff: (line 70, 120, steps: 32).mirror.look, amp: 0.4 if spread(11, 16).look
      sleep 0.25
    end
  end

  # the city: a long dark chord, and something broken speaking now and then
  live_loop :city do
    synth :dark_ambience, note: chord(:d2, :m9), attack: 4, release: 8, amp: 0.7
    sample (ring :glitch_robot1, :glitch_robot2).choose, rate: rrand(0.5, 1), amp: 0.5 if one_in(2)
    sleep 8
  end
end

Algorithmic MIDI

In a DAW you draw MIDI notes into a piano roll. In Sonic Pi you write simple yet powerful algorithms to generate MIDI on the fly. Choosing notes at random from a scale, spreading 5 hits over 16 steps, running loops of different lengths against each other - and you can modify any part of it while it plays.

(In this card the keyboard sends MIDI just as your DAW would)

live_loop :keys do
  note, velocity = sync "/midi:daw:1/note_on"
  sample :loop_amen, onset: note, amp: 2 * velocity / 127.0
end

Each key plays one hit of the amen break.

live_loop :keys do
  note, velocity = sync "/midi:daw:1/note_on"
  sample :guit_em9, onset: note, amp: 2 * velocity / 127.0
end

The same code with a guitar sample.

with_fx :reverb, room: 0.8 do
  live_loop :keys do
    note, velocity = sync "/midi:daw:1/note_on"
    use_synth :pretty_bell
    play note, amp: 1.2 * velocity / 127.0
    in_thread do
      [7, 12, 19].each_with_index do |up, i|
        sleep 0.25
        play note + up, amp: 0.9 / (i + 1)
      end
    end
  end
end

Each note is repeated three times, higher and quieter.

with_fx :reverb, room: 0.8 do
  live_loop :keys do
    note, velocity = sync "/midi:daw:1/note_on"
    in_thread do
      use_synth :chiplead
      chord(note, :minor7, num_octaves: 2).each do |n|
        play n, release: 0.2, amp: 1.2
        sleep 0.125
      end
    end
  end
end

Each key plays a minor 7th chord, one note at a time, over two octaves.

with_fx :reverb, room: 0.8 do
  live_loop :keys do
    note, velocity = sync "/midi:daw:1/note_on"
    in_thread do
      use_random_seed note
      use_synth :pretty_bell
      8.times do
        play scale(:c4, :major_pentatonic, num_octaves: 2).choose, release: 0.3, amp: 1.2
        sleep [0.125, 0.125, 0.125, 0.25].choose
      end
    end
  end
end

Each key plays its own eight note melody. A key always plays the same melody.

Algorithmically generate MIDI and send it to other instruments in your DAW

live_loop :bass do
  notes = scale(:e2, :minor_pentatonic)
  16.times do
    midi notes.choose, sustain: 0.1 if spread(5, 16).tick
    sleep 0.25
  end
end

Deterministic randomisation

Random numbers in Sonic Pi are repeatable. A random melody plays the same every time you run it and every time you bounce it. Change the seed to get a different one.

use_random_seed 42   # try another number

8.times do
  play scale(:e3, :minor_pentatonic).choose, release: 0.2
  sleep 0.25
end

Powered by SuperSonic

At the heart of the plugin is SuperSonic - my variant of scsynth, the audio engine behind SuperCollider. For more than two decades scsynth has been the engine of choice for experimental electronic music and it is astonishingly capable - hundreds of building blocks that can be wired into any synth or effect you can imagine.

SuperSonic takes all of that power and reshapes it to run directly inside your DAW. And if you already design sounds in SuperCollider you can bring them with you - load your own scsynth synthdefs and play them from your code just like the built-in synths.

Live Code your DAW

  • Follows your DAW's tempo and bars
  • Starts and stops with your transport - arm a buffer with its sync button.
  • MIDI in from your keyboard and your clips.
  • MIDI out to any other instrument in your session.
  • Stereo audio in - modify your track’s audio live with code
  • Run multiple instances of the plugin simultaneously
  • A bounce of a random part is the same every time by default.
  • Nudge everything earlier or later by milliseconds to sit with your other tracks.
  • VST3 on macOS and Windows, Audio Unit on macOS.
